import os
import zipfile
import shutil

# Extract the zip file
try:
    with zipfile.ZipFile("archive.zip", "r") as zip_ref:
        zip_ref.extractall()
except Exception as Err:
    print(
        f"Make sure `archive.zip` file exist in the root dir and its not corrupted.\nErr: {Err}"
    )
else:
    # Rename the extracted folder ("TrashType_Image_Dataset") to "Data"
    os.rename("garbage_classification", "Data")

    # Make the main Dataset dir
    os.makedirs("Database", exist_ok=True)

    # Moving the dataset
    shutil.move("Data", "Database/Data")

    # Making the cache dir
    os.makedirs("Database/cache", exist_ok=True)

    # Removing temp files and vars
    os.remove("archive.zip")

    # End
    print("Done!")
